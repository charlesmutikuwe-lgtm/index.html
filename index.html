// ───────────────────────────────────────────────────────────────── // VARSITY VISA — GOOGLE APPS SCRIPT FOR LEAD CAPTURE // ───────────────────────────────────────────────────────────────── // HOW TO USE: // 1. Open script.google.com → New project // 2. Paste this entire file // 3. Change SHEET_NAME if needed (default: "Leads") // 4. Click Deploy → New deployment → Web app // - Execute as: Me // - Who has access: Anyone // 5. Copy the deployment URL into your HTML file // ───────────────────────────────────────────────────────────────── const SHEET_NAME = "Leads"; // Name of the tab in your Google Sheet const SPREADSHEET_ID = ""; // Optional: paste your Sheet ID here if you want // to target a specific spreadsheet. // Leave empty to use the spreadsheet the script is bound to. // Column headers — must match the fields sent from the HTML const HEADERS = [ "Timestamp", "First Name", "Surname", "Full Name", "WhatsApp", "Email", "Programme", "Budget", "Qualification", "Timeline", "Preferred Country", "Has IELTS", "Match 1", "Match 2", "Match 3", "Total Matches", "Source" ]; // ─── HANDLE POST (from HTML form) ──────────────────────────────── function doPost(e) { try { // Parse incoming JSON const data = JSON.parse(e.postData.contents); // Get or create the sheet const sheet = getOrCreateSheet(); // Write headers if sheet is empty if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS); formatHeaders(sheet); } // Build the row const row = [ data.Timestamp || new Date().toISOString(), data.FirstName || "", data.Surname || "", (data.FirstName || "") + " " + (data.Surname || ""), data.WhatsApp || "", data.Email || "", data.Programme || "", data.Budget || "", data.Qualification || "", data.Timeline || "", data.PreferredCountry || "", data.HasIELTS || "", data.Match1 || "", data.Match2 || "", data.Match3 || "", data.TotalMatches || 0, "StudyMatch Platform" ]; // Append to sheet sheet.appendRow(row); // Auto-resize columns for readability sheet.autoResizeColumns(1, HEADERS.length); // Return success return ContentService .createTextOutput(JSON.stringify({ status: "success", message: "Lead saved" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { // Log error and return failure response console.error("doPost error:", err.toString()); return ContentService .createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ─── HANDLE GET (test the endpoint in browser) ─────────────────── function doGet(e) { return ContentService .createTextOutput(JSON.stringify({ status: "ok", message: "Varsity Visa Lead Capture is live", timestamp: new Date().toISOString() })) .setMimeType(ContentService.MimeType.JSON); } // ─── HELPER: Get or create the leads sheet ─────────────────────── function getOrCreateSheet() { let ss; if (SPREADSHEET_ID) { ss = SpreadsheetApp.openById(SPREADSHEET_ID); } else { ss = SpreadsheetApp.getActiveSpreadsheet(); } let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } return sheet; } // ─── HELPER: Format header row ─────────────────────────────────── function formatHeaders(sheet) { const headerRange = sheet.getRange(1, 1, 1, HEADERS.length); // Navy blue background, white text, bold headerRange.setBackground("#0b1f4d"); headerRange.setFontColor("#ffffff"); headerRange.setFontWeight("bold"); headerRange.setFontSize(10); // Freeze header row sheet.setFrozenRows(1); // Set row height sheet.setRowHeight(1, 36); } // ─── MANUAL TEST — run this in Apps Script to check it works ───── function testDoPost() { const fakeEvent = { postData: { contents: JSON.stringify({ Timestamp: new Date().toISOString(), FirstName: "Tawanda", Surname: "Moyo", WhatsApp: "+263771234567", Email: "tawanda@example.com", Programme: "Nursing", Budget: "$3,500 – $5,000 / year", Qualification: "A-Level / Matric equivalent", Timeline: "6–12 months", PreferredCountry: "Any", HasIELTS: "No", Match1: "LSMU · Lithuania (92%)", Match2: "Vincent Pol University · Poland (87%)", Match3: "MAHSA University · Malaysia (81%)", TotalMatches: 6 }) } }; const result = doPost(fakeEvent); console.log("Test result:", result.getContent()); }
